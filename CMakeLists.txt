cmake_minimum_required(VERSION 3.20)
project(ecliptic CXX C)

set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
if(NOT CMAKE_BUILD_TYPE)
  set(CMAKE_BUILD_TYPE RelWithDebInfo)
endif()

find_package(PkgConfig REQUIRED)
pkg_check_modules(SDL2 REQUIRED sdl2)
pkg_check_modules(UNICORN REQUIRED unicorn)
pkg_check_modules(SDL2_IMAGE REQUIRED SDL2_image)
find_package(ZLIB REQUIRED)

file(GLOB ECLIPTIC_SOURCES CONFIGURE_DEPENDS src/*.cpp)
add_executable(ecliptic ${ECLIPTIC_SOURCES})
target_include_directories(ecliptic PRIVATE src ${SDL2_INCLUDE_DIRS} ${SDL2_IMAGE_INCLUDE_DIRS} ${UNICORN_INCLUDE_DIRS})
target_link_directories(ecliptic PRIVATE ${SDL2_LIBRARY_DIRS} ${UNICORN_LIBRARY_DIRS})
target_link_libraries(ecliptic PRIVATE ${SDL2_IMAGE_LIBRARIES} ${SDL2_LIBRARIES} ${UNICORN_LIBRARIES} ZLIB::ZLIB)
if(WIN32)
  target_link_libraries(ecliptic PRIVATE winmm)
endif()
target_compile_options(ecliptic PRIVATE -Wall -Wno-unused-parameter -Wno-unused-function)
target_compile_definitions(ecliptic PRIVATE SDL_MAIN_HANDLED)
if(MINGW)
  target_link_options(ecliptic PRIVATE -mconsole)
endif()
